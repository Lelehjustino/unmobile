  static Future<void> exitProtected({String motivo = ''}) async {
    if (!navigationLocked.value) return;
    if (_acaoEmBanco) return;

    bool conectado = await conectarBanco();

    if (conectado) {
      navigationLocked.value = false;
      _bloquearNavegacao = false;
      voltouAoApp.value = false;

      print(' Navegação LIBERADA: $motivo');
    }
  }

  // Metodo para liberar navegacao
  static void volteiApp() async {
    print(' Voltou ao app');

    voltouAoApp.value = true;
    enterProtected(motivo: 'Voltou ao app');

    // Mostra o loading imediatamente se não estiver conectado, para o usuário ver ação na hora
    if (connectionStatus.value == false || _conn == null || !_conn!.connected) {
      _mostrarLoadingReconexao(force: true);
    }

    await conectarBanco();

  // ================= CONEXÃO =================

  static Future<bool> conexaoRealmenteCaiu() async {
    try {
      if (_conn == null || !_conn!.connected) {
        return true;
      }
      // Timeout agressivo para detectar queda rapidamente
      await _conn!
          .execute('SELECT 1')
          .timeout(const Duration(milliseconds: 400));
      return false;
    } catch (_) {
      return true;
    }
  }

  static Future<MySQLConnection?> getConnection() async {
    if (_isConnecting) return _conn;

    _isConnecting = true;

    try {
      if (_conn != null && _conn!.connected) {
        _safeSetConnectionState(true);
        return _conn;
      }

      if (_falhasSeguidas >= maxFalhasPermitidas) {
        _safeSetConnectionState(false);
        _handleConnectionLost();
        return null;
      }

      final prefs = await SharedPreferences.getInstance();
      String? servidor = prefs.getString('servidor');

      if (servidor != null) {
        bool success = await _tryConnect(servidor);
        if (success) {
          _fecharLoadingReconexao();
          _falhasSeguidas = 0;
          _safeSetConnectionState(true);
          _iniciarKeepAlive();
          return _conn;
        }
      }

      print("Falha conexão");
      _falhasSeguidas++;
      _mostrarLoadingReconexao(force: true);
      await _closeConnSilently();
      // LE
      //_mostrarLoadingReconexao();
      _safeSetConnectionState(false);
      _handleConnectionLost();
      return null;
    } finally {
      _isConnecting = false;
    }
  }

  static Future<bool> conectarBanco() async {
    final conn = await getConnection();
    return conn?.connected ?? false;
  }

  static Future<bool> conectarBanconoSalvar() async {
    voltouAoApp.value = false;
    final prefs = await SharedPreferences.getInstance();
    final servidor = prefs.getString('servidor');
    if (servidor == null) return false;
    // Fecha a conexão antiga e reseta falhas para permitir reconexão limpa
    await _closeConnSilently();
    _falhasSeguidas = 0;
    bool success = await _tryConnect(servidor);
    _safeSetConnectionState(success);
    return success;
  }

  // Funcao chamada quando algo deu errado na conexao
  static void _handleConnectionLost() {
    // PEga o estado atual do app
    final state = WidgetsBinding.instance.lifecycleState;

    if (state == AppLifecycleState.paused) return;

    connectionStatus.value = false;

    // Tenta mostrar o loading (as regras de rota estão dentro do método)
    _mostrarLoadingReconexao(force: true);

    _iniciarReconexaoLoop();
  }

  static void _iniciarReconexaoLoop() async {
    if (_estaReconectando) return;

    _estaReconectando = true;
    int tentativas = 0;

    while (tentativas < 10) {
      final state = WidgetsBinding.instance.lifecycleState;

      if (state == AppLifecycleState.paused) {
        _estaReconectando = false;
        return;
      }

      _falhasSeguidas = 0; // Reseta para permitir nova tentativa
      bool conectado = await conectarBanconoSalvar();

      if (conectado) {
        print(' Reconectado');

        _fecharLoadingReconexao();

        _estaReconectando = false;
        return;
      }

      tentativas++;
      // MUDEI
      // await Future.delayed(const Duration(seconds: 2));
      await Future.delayed(const Duration(milliseconds: 800));
    }

    print('Falha reconexão após 10 tentativas');
    _estaReconectando = false;
  }

  // ================= KEEP ALIVE =================

  static void _iniciarKeepAlive() {
    _keepAliveTimer?.cancel();

    _keepAliveTimer = Timer.periodic(_keepAliveInterval, (timer) async {
      if (_conn == null || !_conn!.connected) {
        _safeSetConnectionState(false);
        _handleConnectionLost();
        timer.cancel();
        return;
      }

      try {
        await _conn!.execute('SELECT 1');
        _safeSetConnectionState(true);
      } catch (_) {
        _safeSetConnectionState(false);
        _handleConnectionLost();
      }
    });
  }

  // ================= BANCO =================

  /// Garante conexão ativa. Se não tiver, mostra loading (exceto /inicial)
  /// e fica tentando reconectar até conseguir ou app sair de foco.
  static Future<MySQLConnection?> _garantirConexao() async {
    // Reage NA HORA! Se já perdemos a conexão em background, não espera
    // o timeout gigante do MySQL para exibir a interface bloqueante.
    if (_conn == null || !_conn!.connected || !connectionStatus.value) {
      _mostrarLoadingReconexao(force: true);
    }

    // 1) Tenta obter conexão normalmente
    MySQLConnection? conn = await getConnection();
    if (conn != null && conn.connected) {
      _fecharLoadingReconexao(); // Garante o fechamento imediato se conectou
      return conn;
    }

    // 2) Sem conexão — tenta mostrar o loading genérico
    _mostrarLoadingReconexao();

    // 3) Loop de reconexão até conseguir
    while (true) {
      await Future.delayed(const Duration(seconds: 1));

      // Se o app saiu de foco, aborta
      final state = WidgetsBinding.instance.lifecycleState;
      if (state == AppLifecycleState.paused) {
        return null;
      }

      _falhasSeguidas = 0; // Reseta para permitir nova tentativa
      bool conectado = await conectarBanconoSalvar();

      if (conectado) {
        _fecharLoadingReconexao();
        connectionStatus.value = true;
        return _conn;
      }
    }
  }

  Future<String> select(String sql, {bool checkLock = true}) async {
    try {
      Connection._acaoEmBanco = true;

      final conn = await _garantirConexao();
      if (conn == null) return "[]";

      final result = await conn.execute(sql);
      return jsonEncode(result.rows.map((r) => r.assoc()).toList());
    } catch (e) {
      print('Erro no SELECT: $e');

      // Verifica uma única vez se a conexão realmente caiu
      if (await Connection.conexaoRealmenteCaiu()) {
        await _closeConnSilently();
        _safeSetConnectionState(false);
        _mostrarLoadingReconexao(force: true);
        _handleConnectionLost();

        // Tenta reconectar e executar de novo UMA vez
        final conn2 = await _garantirConexao();
        if (conn2 != null) {
          try {
            final result = await conn2.execute(sql);
            return jsonEncode(result.rows.map((r) => r.assoc()).toList());
          } catch (_) {}
        }
      }

      return "[]";
    } finally {
      Connection._acaoEmBanco = false;
    }
  }

  Future<String> ipd(String sql, {bool checkLock = true}) async {
    try {
      Connection._acaoEmBanco = true;

  static void _fecharLoadingReconexao() {
    if (!_exibindoCarregamento) return;

    _podeFecharLoading = true;

    if (Get.isDialogOpen ?? false) {
      Get.back();
    }

    _exibindoCarregamento = false;
  }

  static Future<bool> _tryConnect(String host) async {
    if ((host == ipTeste) || (host == ipTeste2)) {
      return await conexaoTeste(host);
    } else {
      return await conexaoProducao(host);
    }
  }

  static Future<bool> conexaoProducao(String host) async {
    try {
      await _closeConnSilently();

      _conn = await MySQLConnection.createConnection(
        host: host,
        port: int.parse(dotenv.get('DB_PORT', fallback: '3306')),
        userName: dotenv.get('DB_USER', fallback: 'root'),
        password: dotenv.get('DB_PASSWORD', fallback: ''),
        databaseName: dotenv.get('DB_NAME', fallback: 'undercontrol'),
        secure: false,
      );

      await _conn!.connect().timeout(const Duration(seconds: 2));

      print('Conexão estabelecida com sucesso!');
      return true;
    } catch (e) {
      print(' Erro ao conectar: $e');
      return false;
    }
  }

  static void _safeSetConnectionState(bool state) {
    try {
      SchedulerBinding.instance.addPostFrameCallback((_) {
        _isConnected.value = state;
      });
    } catch (_) {}
  }

  static Future<void> _closeConnSilently() async {
    _keepAliveTimer?.cancel();
    try {
      await _conn?.close();
    } catch (_) {}
    _conn = null;
  }

  static Future<String?> discoverServer() async {
    const int portaOuvir = 30078;
    const String chaveEsperada = "UnderControl Engine";

    RawDatagramSocket? socket;

    try {
      // No iOS, bind para anyIPv4 é padrão, mas garantimos que o timeout seja maior
      socket = await RawDatagramSocket.bind(
        InternetAddress.anyIPv4,
        portaOuvir,
      );

      socket.broadcastEnabled = true;

      // Aumentamos levemente o timeout no iOS para dar tempo da interface acordar
      final timeoutDuration =
          /* await _conn!.connect().timeout(
        Platform.isIOS 
          ? const Duration(milliseconds: 800)
          : const Duration(seconds: 2)
      );*/
          Platform.isIOS
          ? const Duration(milliseconds: 800)
          : const Duration(seconds: 1);

      await for (RawSocketEvent event in socket.timeout(timeoutDuration)) {
        if (event == RawSocketEvent.read) {
          Datagram? datagram = socket.receive();
          if (datagram == null) continue;

          final String msg = utf8.decode(datagram.data, allowMalformed: true);
          final List<String> split = msg.split('|');

          if (split.length == 2 && split[0] == chaveEsperada) {
            socket.close();
            return split[1].trim();
          }
        }
      }
    } catch (e) {
      print("Erro no discoverServer: $e");
    } finally {
      socket?.close();
    }

    return null;
  }

// ================= LIFECYCLE =================

class _LifecycleHandler with WidgetsBindingObserver {
  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.paused) {
      Connection.enterProtected(motivo: 'App minimizado');
    }

    if (state == AppLifecycleState.resumed) {
      Connection.volteiApp();
    }
  }
}
