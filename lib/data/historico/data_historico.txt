  static Future<String?> discoverServer() async {
    const int portaOuvir = 30078;
    const String chaveEsperada = "UnderControl Engine";
    RawDatagramSocket? socket;
    try {
      socket = await RawDatagramSocket.bind(
        InternetAddress.anyIPv4,
        portaOuvir,
      );
      socket.broadcastEnabled = true;
      await for (RawSocketEvent event in socket.timeout(
        const Duration(seconds: 2),
      )) {
        if (event == RawSocketEvent.read) {
          Datagram? datagram = socket.receive();
          if (datagram == null) continue;
          final String msg = ascii.decode(datagram.data);
          final List<String> split = msg.split('|');
          if (split.length == 2 && split[0] == chaveEsperada) {
            socket.close();
            return split[1];
          }
        }
      }
    } catch (_) {
    } finally {
      socket?.close();
    }
    return null;
  }

  static Future<MySQLConnection?> _garantirConexao(bool checkLock) async {
    MySQLConnection? conn = await getConnection();
    if (conn != null && conn.connected) return conn;

    // Se falhou, avisa o sistema que perdeu conexão
    _handleConnectionLost(acaoDoUsuario: checkLock);

    // Se não for pra travar (checkLock = false), retorna logo
    if (!checkLock) return null;

    // Loop até conectar
    print('[_garantirConexao] Entrando no loop de espera por conexão...');
    while (true) {
      await Future.delayed(const Duration(seconds: 2));

      // Se o app não estiver em primeiro plano, abortamos para não travar infinitamente
      if (WidgetsBinding.instance.lifecycleState != AppLifecycleState.resumed) {
        print('[_garantirConexao] Abortando loop: App fora de foco.');
        return null;
      }

      conn = await getConnection();
      if (conn != null && conn.connected) {
        print(
          '[_garantirConexao] Conexão obtida no loop! Chamando _hideLoading.',
        );
        connectionStatus.value = true;
        _hideLoading();
        return conn;
      }

      // Reforça o sinal de carregamento se for uma ação do usuário
      print(
        '[_garantirConexao] Ainda sem conexão. Reforçando _handleConnectionLost...',
      );
      _handleConnectionLost(acaoDoUsuario: checkLock);
    }
  }

  static Future<String?> discoverServer() async {
    const int portaOuvir = 30078;
    const String chaveEsperada = "UnderControl Engine";
    RawDatagramSocket? socket;
    try {
      socket = await RawDatagramSocket.bind(InternetAddress.anyIPv4, portaOuvir);
      socket.broadcastEnabled = true;
      await for (RawSocketEvent event in socket.timeout(const Duration(seconds: 2))) {
        if (event == RawSocketEvent.read) {
          Datagram? datagram = socket.receive();
          if (datagram == null) continue;
          final String msg = ascii.decode(datagram.data);
          final List<String> split = msg.split('|');
          if (split.length == 2 && split[0] == chaveEsperada) {
            socket.close();
            return split[1];
          }
        }
      }
    } catch (_) {
    } finally {
      socket?.close();
    }
    return null;
  }

  static void _handleConnectionLost({bool acaoDoUsuario = false}) async {
    atualizarBloqueio();

    if (!acaoDoUsuario || !_acaoEmBanco) {
      print('Ignorado: não foi ação direta do usuário');
      return;
    }

    if (_bloquearNavegacao) return;

    bool caiu = await conexaoRealmenteCaiu();

    // Se a conexão AINDA ESTÁ VIVA, NÃO navega
    if (!caiu) {
      print('Erro SQL, mas conexão ativa. NÃO navegando.');
      return;
    }

    if (WidgetsBinding.instance.lifecycleState == AppLifecycleState.resumed) {
      await Future.delayed(const Duration(milliseconds: 300));
      if (!_bloquearNavegacao) {
        connectionStatus.value = false;
        // Quando em direto do fechamento do aplicativo, não navega 
        if(!voltouAoApp.value){
          print('Conexão realmente perdida. Navegando para /erro');
          Get.offAllNamed('/erro');
          zerarAtendimento.value = true;
        }
      }
    }
  }
